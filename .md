  // física para que a paginação (quantos px cabem numa folha) seja
  // consistente independentemente do zoom/ajuste à tela.
  const PAGE_W = 794;

  const PAGE_PAD_Y = 76;
  const PAGE_PAD_X = 64;
  const CONTENT_H = PAGE_H - PAGE_PAD_Y * 2;

  let containerEl;
  let fitScale = 1;
  let pinchScale = 1;

  function computeFitScale() {
    if (!containerEl) return;
    const availableWidth = containerEl.clientWidth - 32;
    fitScale = Math.min(1, availableWidth / PAGE_W);
    if (fitScale <= 0 || !isFinite(fitScale)) fitScale = 1;

  //  PAGINAÇÃO REAL — um único contenteditable "master" (invisível,
  //  fora do ecrã) recebe toda a digitação/seleção nativa do browser.
  //  A cada mutação, o conteúdo do master é fatiado em blocos que cabem
  //  em CONTENT_H de altura e distribuído por N folhas A4 visuais
  //  (não-editáveis diretamente — cada folha é só uma "janela" que
  //  mostra a fatia correspondente do master via um clone). Isto imita
  //  o comportamento do Word/Microsoft 365: ao encher uma folha, o
  //  conteúdo excedente "cai" automaticamente para a folha seguinte.
  //
  //  Para manter 100% a fidelidade do cursor nativo do browser (sem
  //  reimplementar seleção), o master é o ÚNICO elemento realmente
  //  editável e ocupa a folha "ativa" (a que contém o cursor); as
  //  restantes folhas mostram snapshots estáticos (innerHTML clonado)
  //  do que "sobrou". Ao clicar numa folha não-ativa, o master é
  //  fisicamente movido para dentro dela antes do foco (preserva
  //  exatamente a posição visual/scroll).

    return el.scrollHeight - CONTENT_H;

  // início do elemento `to`, até `from` deixar de estourar CONTENT_H.
  function pullOverflowingNodes(from, to) {
    let guard = 0;
    while (measureOverflow(from) > 0 && from.lastElementChild && guard < 2000) {
      guard++;
      const last = from.lastElementChild;
      // Se o próprio nó sozinho já é maior que a página (ex: imagem
      // grande), ainda assim movemos - a próxima página absorve.
      to.insertBefore(last, to.firstChild);
      if (measureOverflow(from) <= 0 && to.firstElementChild) {
        // pode ter movido demais quando um único nó pequeno resolvia;
        // não há forma barata de "meio-nó" com blocos, então paramos
        // aqui — é o mesmo comportamento de quebra por parágrafo do Word.
      }
    }

  function pullUnderflowingNodes(from, to) {

    // ainda tem conteúdo a menos que caiba, puxa de volta o primeiro
    // nó de `to` para `from`, enquanto ainda houver espaço.
    let guard = 0;
    while (to.firstElementChild && guard < 2000) {

    if (containerEl) resizeObserver.observe(containerEl);
    window.addEventListener('orientationchange', computeFitScale);

    // Cria o master e injeta o conteúdo inicial
    masterEl.innerHTML = initialContent || '<p><br></p>';
    tick().then(() => {
      mountMasterIntoActivePage();
      dispatch('ready', { html: getContent() });
      repaginate();
    });

    window.addEventListener('mousemove', onDocPointerMove);
    window.addEventListener('mouseup', onDocPointerUp);
    window.addEventListener('touchmove', onDocPointerMove, { passive: false });
    window.addEventListener('touchend', onDocPointerUp);
  });
  onDestroy(() => {
    resizeObserver?.disconnect();
    window.removeEventListener('orientationchange', computeFitScale);
    window.removeEventListener('mousemove', onDocPointerMove);
    window.removeEventListener('mouseup', onDocPointerUp);
    window.removeEventListener('touchmove', onDocPointerMove);
    window.removeEventListener('touchend', onDocPointerUp);
    if (repaginateRaf) cancelAnimationFrame(repaginateRaf);
    document.getElementById('__doc_scratch__')?.remove();
  });

  $: effectiveScale = fitScale * pinchScale;
</script>
